using Eagle.BaseGame;
using UnityEngine;

public class BaseTimeScaleUI : BaseUI
{
    protected override void OnEnable()
    {
        base.OnEnable();
        Time.timeScale = 0;
    }

    protected override void Invisible()
    {
        base.Invisible();

        var uiOnTop = GetUIOnTop();
        if (uiOnTop == null)
        {
            Time.timeScale = 1;
            return;
        }

        var baseTimeScaleUI = uiOnTop.GetComponent<BaseTimeScaleUI>();
        if (baseTimeScaleUI == null)
            Time.timeScale = 1;
    }

    private GameObject GetUIOnTop()
    {
        var mainCanvas = UIManager.Instance.MainCanvas;
        for (var i = mainCanvas.childCount - 1; i >= 0; i--)
        {
            var child = mainCanvas.GetChild(i).gameObject;
            if (child.activeInHierarchy)
                return child;
        }
        return null;
    }
}
